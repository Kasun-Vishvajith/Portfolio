% ── PROJECTS ──────────────────────────────────────────────────
\section{Projects}

\cvproject{Diamond Intel}{Python, CatBoost}
\begin{cvlist}
  \item Solved the gemstone valuation problem where highly nonlinear relationships make pricing complex and unstable.
  \item Trained a CatBoost regression model using log transformed diamond prices and physical attributes (carat, cut, colour, clarity, dimensions).
  \item Built a web application that outputs statistically bounded price ranges with 90\% and 95\% confidence intervals using a two-tier model system.
  \item Gained insight into how log transformation stabilizes skewed targets and improves model generalization in regression tasks.
\end{cvlist}

\begin{samepage}
\cvproject{EV Range Analysis and EDA}{R, Multiple Linear Regression, Logistic Regression}
\begin{cvlist}
  \item Conducted exploratory and predictive analysis on a 478 observation electric vehicle dataset to understand factors influencing driving range.
  \item Performed extensive EDA using scatter plots, box plots, and categorical comparisons across EV segment, drivetrain, and physical specifications.
  \item Built forward selection multiple linear regression achieving an Adjusted $R^2$ of 0.952 for range prediction.
  \item Developed a logistic regression model achieving 95.8\% accuracy in classifying EVs into short range and long range categories.
  \item Identified key patterns such as larger and wider vehicles being associated with higher range, while taller vehicles tend to show reduced range.
  \item Learned how EDA complements regression by revealing relationships not fully captured by model coefficients alone.
\end{cvlist}
\end{samepage}

\cvproject{Tea Yield Forecasting}{Python, SARIMAX, CatBoost}
\begin{cvlist}
  \item Forecasted Sri Lankan tea production across 21 combinations of tea types and elevation levels.
  \item Developed a SARIMAX model to capture overall seasonal production trends.
  \item Built separate CatBoost models for each tea type and elevation combination using environmental variables (rainfall, temperature, humidity).
  \item Generated category specific forecasts that reflect environmental and elevation driven variability in production.
  \item Learned how decomposing a multi output forecasting problem improves interpretability and model performance.
\end{cvlist}
